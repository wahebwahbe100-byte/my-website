const typingElement = document.getElementById("typingText"); const typingLines = [ { plain: 'name: "Modern Website",', html: 'name: "Modern Website",' }, { plain: 'speed: "Fast",', html: 'speed: "Fast",' }, { plain: 'responsive: true,', html: 'responsive: true,' }, { plain: 'seo: "Optimized",', html: 'seo: "Optimized",' }, { plain: 'client: "Happy",', html: 'client: "Happy",' } ]; let typingLineIndex = 0; let typingCharIndex = 0; let deleting = false; function renderPartialHTML(fullHTML, visibleChars) { const wrapper = document.createElement("div"); wrapper.innerHTML = fullHTML; let count = 0; function walk(node) { if (node.nodeType === Node.TEXT_NODE) { const text = node.textContent || ""; if (count >= visibleChars) { node.textContent = ""; } else if (count + text.length > visibleChars) { node.textContent = text.slice(0, visibleChars - count); count = visibleChars; } else { count += text.length; } } else if (node.nodeType === Node.ELEMENT_NODE) { [...node.childNodes].forEach(walk); } } [...wrapper.childNodes].forEach(walk); return wrapper.innerHTML; } function runTypingEffect() { if (!typingElement) return; const current = typingLines[typingLineIndex]; const plainText = current.plain; if (!deleting) { typingCharIndex++; typingElement.innerHTML = renderPartialHTML(current.html, typingCharIndex); if (typingCharIndex >= plainText.length) { deleting = true; setTimeout(runTypingEffect, 1200); return; } } else { typingCharIndex--; typingElement.innerHTML = renderPartialHTML(current.html, typingCharIndex); if (typingCharIndex <= 0) { deleting = false; typingLineIndex = (typingLineIndex + 1) % typingLines.length; } } setTimeout(runTypingEffect, deleting ? 45 : 85); } runTypingEffect();
